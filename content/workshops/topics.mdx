    audience: "Engineering · QA"

  - title: "Prompt caching, model fallback, observability"
    audience: "Engineering · Platform"
    summary: >

  - title: "Workflows over chatbots — for non-technical teams"
    audience: "Operators · Sales · Marketing"
    summary: >
      A non-code session for teams adopting AI without engineering support.

    audience: "Engineering"

    audience: "ML · Engineering · Product"
    summary: >
